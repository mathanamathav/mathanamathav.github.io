{{- /* Markdown image render hook: convert bundle images to WebP, make them responsive, lazy-load off-screen ones, and emit width/height to avoid CLS. */ -}} {{- $img := .Page.Resources.GetMatch .Destination -}} {{- if $img -}} {{- $max := 1100 -}} {{- $tw := $img.Width -}} {{- if gt $tw $max -}}{{- $tw = $max -}}{{- end -}} {{- $full := $img.Resize (printf "%dx webp q90" $tw) -}} {{- $half := $img.Resize (printf "%dx webp q90" (div $tw 2)) -}} {{- else -}} {{- end -}}
